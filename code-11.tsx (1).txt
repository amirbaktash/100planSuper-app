    { href: "/admin/tours",    label: "تورها",      need: "tours:EDIT" },      // ★
    { href: "/admin/policies", label: "بیمه",       need: "policies:VIEW" },   // ★
    { href: "/admin/media",    label: "گالری",      need: "cms:EDIT" },        // ★

// WYSIWYG: Tiptap editor + MediaPicker sidebar
export function PageEditor({ page }: { page: any }) {
  const editor = useEditor({
    extensions: [StarterKit, Image.configure({ inline: true })],
    content: page.blocks });
  return (
    <div className="flex gap-4">
      <div className="flex-1 border rounded p-4" dir="rtl">
        <EditorContent editor={editor} />
        <button onClick={() => save({ blocks: editor?.getJSON() }, /* → PENDING */)}
          className="mt-4 px-4 py-2 bg-blue-600 text-white rounded">ثبت برای تأیید</button>
      </div>
      <MediaPicker tag="banner" onSelect={url => editor?.chain().focus().setImage({ src: url }).run()} />
    </div>);
}
